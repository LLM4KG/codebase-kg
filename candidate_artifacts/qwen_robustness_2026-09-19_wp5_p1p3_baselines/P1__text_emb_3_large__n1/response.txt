  const decreaseProductQuantity = (productToDecrease: ICartProduct) => {
    const updatedProducts = products.reduce((acc: ICartProduct[], product: ICartProduct) => {
      if (product.id === productToDecrease.id) {
        const newQuantity = product.quantity - 1;
        if (newQuantity <= 0) {
          // Remove product from cart if quantity would be zero or less
          return acc;
        } else {
          // Update quantity if still positive
          return [...acc, { ...product, quantity: newQuantity }];
        }
      } else {
        // Keep other products unchanged
        return [...acc, product];
      }
    }, []);

    setProducts(updatedProducts);
    updateCartTotal(updatedProducts);
