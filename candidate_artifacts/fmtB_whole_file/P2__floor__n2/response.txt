src/client/slices/note.ts
```ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit'
import { NoteItem, NoteState } from '@/types'
import { getRehydratedNotes } from '@/utils/rehydration'

const initialState: NoteState = getRehydratedNotes()

const noteSlice = createSlice({
  name: 'note',
  initialState,
  reducers: {
    addNote: (state, action: PayloadAction<NoteItem>) => {
      state.notes = [action.payload, ...state.notes]
    },
    updateNote: (state, action: PayloadAction<NoteItem>) => {
      state.notes = state.notes.map(note =>
        note.id === action.payload.id ? action.payload : note
      )
    },
    deleteNote: (state, action: PayloadAction<string>) => {
      state.notes = state.notes.filter(note => note.id !== action.payload)
    },
    swapNote: (state, action: PayloadAction<string>) => {
      state.activeNoteId = action.payload
    },
    swapCategory: (state, action: PayloadAction<string>) => {
      state.activeCategoryId = action.payload
    },
    swapLinkedCategory: (state, action: PayloadAction<string>) => {
      state.activeCategoryId = action.payload
    },
    toggleFavoriteNote: (state, action: PayloadAction<string>) => {
      state.notes = state.notes.map(note =>
        note.id === action.payload ? { ...note, favorite: !note.favorite } : note
      )
    },
    toggleTrashedNote: (state, action: PayloadAction<string>) => {
      state.notes = state.notes.map(note =>
        note.id === action.payload ? { ...note, trash: !note.trash } : note
      )
    },
    restoreAllTrash: state => {
      state.notes = state.notes.map(note =>
        note.trash ? { ...note, trash: false } : note
      )
    },
    emptyTrash: state => {
      state.notes = state.notes.filter(note => !note.trash)
    },
    updateActiveNote: (state, action: PayloadAction<string>) => {
      state.activeNoteId = action.payload
    },
    loadNotes: (state, action: PayloadAction<NoteItem[]>) => {
      state.notes = action.payload
    },
    pruneNotes: state => {
      state.notes = state.notes.filter(note => note.text !== '')
    },
  },
})

export const {
  addNote,
  updateNote,
  deleteNote,
  swapNote,
  swapCategory,
  swapLinkedCategory,
  toggleFavoriteNote,
  toggleTrashedNote,
  restoreAllTrash,
  emptyTrash,
  updateActiveNote,
  loadNotes,
  pruneNotes,
} = noteSlice.actions

export default noteSlice.reducer

import React, { useState, useRef, useEffect } from 'react'
import { useDispatch, useSelector } from 'react-redux'
import { Droppable } from 'react-beautiful-dnd'
import { MoreHorizontal, Trash, RefreshCw } from 'react-feather'

import NoteItem from '@/containers/NoteItem'
import { TestID } from '@/resources/TestID'
import { Folder } from '@/utils/enums'
import { getNoteTitle, sortByLastUpdated } from '@/utils/helpers'
import { emptyTrash, restoreAllTrash } from '@/slices/note'
import {
  selectActiveCategoryId,
  selectActiveFolder,
  selectActiveNoteId,
  selectNotes,
} from '@/selectors'

const NoteList: React.FC = () => {
  const dispatch = useDispatch()

  const activeFolder = useSelector(selectActiveFolder)
  const activeCategoryId = useSelector(selectActiveCategoryId)
  const activeNoteId = useSelector(selectActiveNoteId)
  const notes = useSelector(selectNotes)

  const [noteOptionsId, setNoteOptionsId] = useState('')
  const node = useRef<HTMLDivElement>(null)

  const isTrash = activeFolder === Folder.TRASH

  const filteredNotes: any[] = notes
    .filter(note => {
      if (activeFolder === Folder.FAVORITES) {
        return !note.trash && note.favorite
      } else if (activeFolder === Folder.TRASH) {
        return note.trash
      } else if (activeFolder === Folder.CATEGORY) {
        return !note.trash && note.category === activeCategoryId
      } else {
        return !note.trash
      }
    })
    .sort(sortByLastUpdated)

  const handleDomClick = (event: MouseEvent) => {
    const element = event.target as HTMLDivElement
    if (node.current && node.current.contains(element)) return
    setNoteOptionsId('')
  }

  useEffect(() => {
    document.addEventListener('mousedown', handleDomClick)
    return () => {
      document.removeEventListener('mousedown', handleDomClick)
    }
  })

  return (
    <aside className="note-sidebar">
      <div className="note-sidebar-header" data-testid={TestID.NOTE_SIDEBAR}>
        {isTrash && (
          <div className="trash-actions">
            <button
              className="button-restore-all"
              data-testid={TestID.RESTORE_ALL_TRASH}
              onClick={() => dispatch(restoreAllTrash())}
            >
              <RefreshCw size={14} className="button-icon" />
              Restore All
            </button>
            <button
              className="button-empty-trash"
              data-testid={TestID.EMPTY_TRASH}
              onClick={() => dispatch(emptyTrash())}
            >
              <Trash size={14} className="button-icon" />
              Empty Trash
            </button>
          </div>
        )}
      </div>
      <Droppable droppableId="note-list">
        {provided => (
          <div
            {...provided.droppableProps}
            ref={provided.innerRef}
            className="note-list"
          >
            {filteredNotes.map((note, index) => (
              <NoteItem
                key={note.id}
                note={note}
                index={index}
                title={getNoteTitle(note.text)}
                active={note.id === activeNoteId}
                noteOptionsId={noteOptionsId}
                setNoteOptionsId={setNoteOptionsId}
              />
            ))}
            {provided.placeholder}
          </div>
        )}
      </Droppable>
    </aside>
  )
}

export default NoteList
