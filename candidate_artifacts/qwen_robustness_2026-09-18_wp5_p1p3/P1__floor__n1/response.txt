  const decreaseProductQuantity = (productId: string) => {
    setCartProducts(prev => {
      const product = prev.find(p => p.id === productId);
      if (!product) return prev;
      
      const newQuantity = product.quantity - 1;
      if (newQuantity <= 0) {
        return prev.filter(p => p.id !== productId);
      }
