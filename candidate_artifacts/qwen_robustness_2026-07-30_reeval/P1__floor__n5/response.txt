  const decreaseProductQuantity = (productId: string, quantity: number) => {
    setCartProducts(prevCartProducts => {
      return prevCartProducts.reduce((acc, cartProduct) => {
        if (cartProduct.product.id === productId) {
          const newQuantity = cartProduct.quantity - quantity;
          if (newQuantity > 0) {
            acc.push({
              ...cartProduct,
              quantity: newQuantity
            });
          }
          // If newQuantity is <= 0, we simply don't add it back to the cart

          acc.push(cartProduct);
