    addCategory: (state, { payload }: PayloadAction<CategoryItem>) => {
      const categoryName = payload.name.toLowerCase()
      const categoryExists = state.categories.some(
        (category) => category.name.toLowerCase() === categoryName
      )

      if (!categoryExists) {
        state.categories.push(payload)
      }
    },

    importCategories: (state, { payload }: PayloadAction<CategoryItem[]>) => {
      const categoryNames = new Map<string, string>()
      state.categories.forEach(({ name }) => categoryNames.set(name.toLowerCase(), name))

      // Make sure duplicate category is not added
      const toAdd = payload.filter(({ name }) => !categoryNames.has(name.toLowerCase()))

      state.categories.push(...toAdd)
    },

    updateCategory: (state, { payload }: PayloadAction<CategoryItem>) => {

      )

        state.categories = state.categories.map((category) =>
