import useCartTotal from './useCartTotal';

const useCart = () => {
  const { isOpen, setIsOpen } = useCartContext();

    products,

    removeProduct,

    decreaseProductQuantity,

  } = useCartProducts();

  const openCart = () => setIsOpen(true);

  };
};

export default useCart;
