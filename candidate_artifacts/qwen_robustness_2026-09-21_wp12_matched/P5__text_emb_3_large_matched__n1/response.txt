import CartProduct from './CartProduct';

import * as S from './style';
import { ICartProduct } from 'models';
import { useCart } from 'contexts/cart-context';

const CartProducts = () => {
  const { products } = useCart();
  const { removeProduct, increaseProductQuantity, decreaseProductQuantity } = useCart();

  return (
    <S.Container>
      {products.length > 0 ? (
        products.map((product: ICartProduct) => (
          <CartProduct 
            key={product.sku} 
            product={product}
            onRemove={removeProduct}
            onIncrease={increaseProductQuantity}
            onDecrease={decreaseProductQuantity}
          />
        ))
      ) : (
        <S.CartEmpty>
          <S.CartEmptyText>Your cart is empty</S.CartEmptyText>
        </S.CartEmpty>
      )}
    </S.Container>
  );
};

export default CartProducts;
